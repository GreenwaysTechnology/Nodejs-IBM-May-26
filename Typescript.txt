function PrintId(id: string | number | undefined) {
    //console.log(id.toUpperCase())
    if (typeof id === "string") {
        // In this branch, id is of type 'string'
        console.log(id.toUpperCase());
    } else if (typeof id === "number") {
        // Here, id is of type 'number'
        console.log(id.toFixed());
    } else {
        console.log(id)
    }
}
